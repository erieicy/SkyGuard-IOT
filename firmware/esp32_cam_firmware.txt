#include "esp_camera.h"
#include <WiFi.h>
#include <HTTPClient.h>
#include <Preferences.h>

// ==========================================
// 1. PENGATURAN WI-FI & SERVER
// ==========================================
const char* WIFI_SSID     = "NAMA_WIFI_ANDA";
const char* WIFI_PASSWORD = "PASSWORD_WIFI_ANDA";

// URL Dasar Server SkyGuard AI (bisa diubah via Dashboard "Hubungkan")
// Contoh: "http://192.168.1.100/SkyGuard-AI/api/esp32.php"
String baseServerUrl = "http://192.168.1.100/SkyGuard-AI/api/esp32.php";

// URL Endpoint dibentuk dari baseServerUrl
String SERVER_UPLOAD_URL   = baseServerUrl + "?action=upload_cam";
String SERVER_COMMAND_URL  = baseServerUrl + "?action=get_command";

// Web server kecil untuk menerima perintah "Hubungkan" dari Dashboard
WiFiServer esp32CamServer(80);
Preferences camPreferences;

// ==========================================
// 2. PINOUT CAMERA AI-THINKER ESP32-CAM
// ==========================================
#define PWDN_GPIO_NUM     32
#define RESET_GPIO_NUM    -1
#define XCLK_GPIO_NUM      0
#define SIOD_GPIO_NUM     26
#define SIOC_GPIO_NUM     27

#define Y9_GPIO_NUM       35
#define Y8_GPIO_NUM       34
#define Y7_GPIO_NUM       39
#define Y6_GPIO_NUM       36
#define Y5_GPIO_NUM       21
#define Y4_GPIO_NUM       19
#define Y3_GPIO_NUM       18
#define Y2_GPIO_NUM        5
#define VSYNC_GPIO_NUM    25
#define HREF_GPIO_NUM     23
#define PCLK_GPIO_NUM     22

#define FLASH_LED_PIN      4  // Onboard High-Brightness Flash LED (kamera, bukan indikator)

// ==========================================
// 2B. PIN AKTUATOR ATAP (2 penggerak: buka & tutup)
//     Tidak ada LED/Buzzer. Atap digerakkan langsung oleh 2 aktuator.
// ==========================================
// Mode deploy:
//   1 = ESP32-CAM LANGSUNG menggerakkan 2 aktuator atap (setup 1 board tunggal).
//   0 = atap digerakkan oleh ESP32 Controller TERPISAH via polling server
//       (lihat esp32_firmware.txt); kamera hanya kirim foto & laporkan verdict.
#define DRIVE_ACTUATORS_FROM_CAM 1

#define ACTUATOR_OPEN_PIN   14  // GPIO penggerak atap MEMBUKA
#define ACTUATOR_CLOSE_PIN  13  // GPIO penggerak atap MENUTUP
// Durasi pulsa aktuator (ms) — sesuaikan dengan waktu buka/tutup fisik atap
const unsigned long ROOF_MOVE_MS = 3000;

// Interval Pengambilan Foto Langit Berkala: tiap 5 menit (300.000 ms).
// Di-reset setiap kali foto DIAMBIL (baik otomatis maupun manual oleh pengguna),
// sehingga jika pengguna memfoto di tengah-tengah timer, hitungan 5 menit mulai dari awal.
const unsigned long CAPTURE_INTERVAL = 300000;
unsigned long lastCaptureTime = 0;

// Interval Polling Perintah Dashboard (setiap 3 detik)
const unsigned long POLL_INTERVAL = 3000;
unsigned long lastPollTime = 0;

// ==========================================
// 3. SETUP SISTEM
// ==========================================
void setup() {
  Serial.begin(115200);
  delay(1000);
  Serial.println("\n===========================================");
  Serial.println("  SKYGUARD AI - ESP32-CAM VISION BOOTING   ");
  Serial.println("===========================================");

  pinMode(FLASH_LED_PIN, OUTPUT);
  digitalWrite(FLASH_LED_PIN, LOW);

  // Inisialisasi 2 pin aktuator atap (buka & tutup)
  setupActuators();

  // Inisialisasi Kamera OV2640
  if (!initCamera()) {
    Serial.println("[ERROR] Inisialisasi Kamera GAGAL! Periksa modul kamera.");
    while (true) { delay(1000); }
  }
  Serial.println("[KAMERA] OV2640 Berhasil Diinisialisasi!");

  // Koneksi Wi-Fi
  connectWiFi();

  // Muat URL Server dari memori persisten (jika diatur via Dashboard "Hubungkan")
  camPreferences.begin("skyguard_cam", false);
  String savedUrl = camPreferences.getString("server_url", "");
  if (savedUrl.length() > 0) {
    baseServerUrl = savedUrl;
    SERVER_UPLOAD_URL = baseServerUrl + "?action=upload_cam";
    SERVER_COMMAND_URL = baseServerUrl + "?action=get_command";
    Serial.println("[CONFIG] URL Server dimuat dari memori: " + baseServerUrl);
  } else {
    Serial.println("[CONFIG] URL Server default: " + baseServerUrl);
  }

  // Mulai Web Server kecil di port 80 untuk menerima perintah dari Dashboard
  esp32CamServer.begin();
  Serial.println("[WEB] ESP32-CAM HTTP Server aktif di port 80. Dashboard dapat mengirim /connect?server=URL");

  // Ambil foto pertama setelah boot
  delay(2000);
  captureAndUploadSkyImage();
}

// ==========================================
// 4. LOOP UTAMA
// ==========================================
void loop() {
  if (WiFi.status() != WL_CONNECTED) {
    connectWiFi();
  }

  // Tangani perintah masuk dari Dashboard (misal: /connect?server=URL)
  handleCamEsp32Server();

  // 1. Cek perintah snapshot instan dari Dashboard
  if (millis() - lastPollTime >= POLL_INTERVAL) {
    lastPollTime = millis();
    checkRemoteSnapshotTrigger();
  }

  // 2. Pengambilan foto berkala otomatis (tiap CAPTURE_INTERVAL)
  //    Timer di-reset di dalam captureAndUploadSkyImage() -> berlaku juga
  //    untuk foto manual dari pengguna (hitungan 5 menit mulai dari awal).
  if (millis() - lastCaptureTime >= CAPTURE_INTERVAL) {
    captureAndUploadSkyImage();
  }

  delay(100);
}

void checkRemoteSnapshotTrigger() {
  if (WiFi.status() != WL_CONNECTED) return;

  HTTPClient http;
  http.begin(SERVER_COMMAND_URL);
  int httpCode = http.GET();
  if (httpCode == HTTP_CODE_OK) {
    String payload = http.getString();
    if (payload.indexOf("\"trigger_snapshot\":true") >= 0) {
      Serial.println("\n[DASHBOARD TRIGGER] Perintah ambil foto diterima dari pengguna!");
      captureAndUploadSkyImage();
    }
  }
  http.end();
}

// ==========================================
// 4B. WEB SERVER KECIL (Terima Perintah "Hubungkan" dari Dashboard)
// ==========================================
void handleCamEsp32Server() {
  WiFiClient client = esp32CamServer.available();
  if (!client) return;

  // Baca baris request pertama: "GET /connect?server=... HTTP/1.1"
  String reqLine = client.readStringUntil('\n');
  reqLine.trim();

  // Ambil path + query string
  int firstSpace = reqLine.indexOf(' ');
  int secondSpace = reqLine.indexOf(' ', firstSpace + 1);
  String path = (firstSpace >= 0 && secondSpace >= 0)
      ? reqLine.substring(firstSpace + 1, secondSpace)
      : "";

  // Baca sisa header hingga baris kosong
  while (client.available()) {
    String line = client.readStringUntil('\n');
    if (line == "\r" || line == "\n" || line == "") break;
  }

  String body = "SkyGuard ESP32-CAM Ready";

  if (path.startsWith("/connect")) {
    // Parse query: ?server=URL_ENCODED
    int q = path.indexOf("server=");
    if (q >= 0) {
      String newUrl = path.substring(q + 7);

      // URL-decode sederhana untuk karakter umum
      newUrl.replace("%2F", "/");
      newUrl.replace("%3A", ":");
      newUrl.replace("%3D", "=");
      newUrl.replace("%26", "&");
      newUrl.replace("+", "");

      baseServerUrl = newUrl;
      SERVER_UPLOAD_URL = baseServerUrl + "?action=upload_cam";
      SERVER_COMMAND_URL = baseServerUrl + "?action=get_command";
      camPreferences.putString("server_url", baseServerUrl);
      Serial.println("\n[WEB] Perintah 'Hubungkan' diterima dari Dashboard.");
      Serial.println("[WEB] ESP32-CAM akan mempolling server: " + baseServerUrl);
      body = "SKYGUARD_OK ESP32-CAM sekarang mempolling: " + baseServerUrl;
    } else {
      body = "ERROR: parameter 'server' tidak ditemukan";
    }
  }

  // Kirim respons HTTP sederhana
  String resp = "HTTP/1.1 200 OK\r\n";
  resp += "Content-Type: text/plain\r\n";
  resp += "Connection: close\r\n\r\n";
  resp += body;
  client.print(resp);
  client.stop();
}

// ==========================================
// 5. INISIALISASI KAMERA
// ==========================================
bool initCamera() {
  camera_config_t config;
  config.ledc_channel = LEDC_CHANNEL_0;
  config.ledc_timer = LEDC_TIMER_0;
  config.pin_d0 = Y2_GPIO_NUM;
  config.pin_d1 = Y3_GPIO_NUM;
  config.pin_d2 = Y4_GPIO_NUM;
  config.pin_d3 = Y5_GPIO_NUM;
  config.pin_d4 = Y6_GPIO_NUM;
  config.pin_d5 = Y7_GPIO_NUM;
  config.pin_d6 = Y8_GPIO_NUM;
  config.pin_d7 = Y9_GPIO_NUM;
  config.pin_xclk = XCLK_GPIO_NUM;
  config.pin_pclk = PCLK_GPIO_NUM;
  config.pin_vsync = VSYNC_GPIO_NUM;
  config.pin_href = HREF_GPIO_NUM;
  config.pin_sscb_sda = SIOD_GPIO_NUM;
  config.pin_sscb_scl = SIOC_GPIO_NUM;
  config.pin_pwdn = PWDN_GPIO_NUM;
  config.pin_reset = RESET_GPIO_NUM;
  config.xclk_freq_hz = 20000000;
  config.pixel_format = PIXFORMAT_JPEG;

  // Sesuaikan resolusi & buffer untuk kestabilan RAM
  if (psramFound()) {
    config.frame_size = FRAMESIZE_VGA; // 640x480 (Ideal untuk AI Vision)
    config.jpeg_quality = 12;          // 0-63, makin kecil kualitas makin tinggi
    config.fb_count = 2;
  } else {
    config.frame_size = FRAMESIZE_QVGA; // 320x240
    config.jpeg_quality = 15;
    config.fb_count = 1;
  }

  esp_err_t err = esp_camera_init(&config);
  return (err == ESP_OK);
}

// ==========================================
// 6. AMBIL FOTO & UPLOAD MULTIPART KE SERVER
// ==========================================
void captureAndUploadSkyImage() {
  if (WiFi.status() != WL_CONNECTED) return;

  Serial.println("\n[KAMERA] Mengambil foto kondisi langit / cuaca...");
  
  camera_fb_t * fb = esp_camera_fb_get();
  if (!fb) {
    Serial.println("[ERROR] Gagal mengambil frame kamera!");
    return;
  }

  Serial.printf("[KAMERA] Foto berhasil diambil! Ukuran buffer: %u bytes\n", fb->len);

  // Buat HTTP Client untuk multipart/form-data
  WiFiClient client;
  HTTPClient http;

  if (http.begin(client, SERVER_UPLOAD_URL)) {
    String boundary = "----WebKitFormBoundarySkyGuardAI" + String(millis());
    http.addHeader("Content-Type", "multipart/form-data; boundary=" + boundary);

    // Siapkan body multipart
    String head = "--" + boundary + "\r\n" +
                  "Content-Disposition: form-data; name=\"source\"\r\n\r\n" +
                  "esp32_cam\r\n" +
                  "--" + boundary + "\r\n" +
                  "Content-Disposition: form-data; name=\"image\"; filename=\"esp32_sky.jpg\"\r\n" +
                  "Content-Type: image/jpeg\r\n\r\n";

    String tail = "\r\n--" + boundary + "--\r\n";

    size_t totalLen = head.length() + fb->len + tail.length();

    // Alokasi buffer gabungan untuk transmisi cepat
    uint8_t * payload = (uint8_t *)malloc(totalLen);
    if (payload != NULL) {
      memcpy(payload, head.c_str(), head.length());
      memcpy(payload + head.length(), fb->buf, fb->len);
      memcpy(payload + head.length() + fb->len, tail.c_str(), tail.length());

      Serial.println("[HTTP UPLOAD] Mengunggah foto ke SkyGuard AI Server...");
      int httpResponseCode = http.POST(payload, totalLen);

      if (httpResponseCode > 0) {
        String response = http.getString();
        Serial.printf("[HTTP UPLOAD] Sukses! Response Code: %d\n", httpResponseCode);
        Serial.println("[AI VERDICT] " + response);
        handleAiVerdict(response);   // Keluarkan perintah TUTUP / BUKA
      } else {
        Serial.printf("[ERROR] Gagal upload foto. HTTP Error: %s\n", http.errorToString(httpResponseCode).c_str());
      }

      free(payload);
    } else {
      Serial.println("[ERROR] Kehabisan memori untuk buffer multipart!");
    }

    http.end();
  }

  esp_camera_fb_return(fb); // Bersihkan buffer kamera

  // Reset timer 5 menit: berlaku untuk foto otomatis MAUPUN foto manual pengguna.
  // Jika pengguna memfoto di tengah-tengah timer, hitungan 5 menit dihitung ulang dari sini.
  lastCaptureTime = millis();
}

// ==========================================
// 6B. PARSE VERDICT AI & KELUARKAN PERINTAH TUTUP / BUKA
// ==========================================
void handleAiVerdict(String response) {
  // Ambil field "roof_action" (atau fallback "new_roof_status") dari JSON respons server.
  String action = extractJsonField(response, "roof_action");
  if (action == "") action = extractJsonField(response, "new_roof_status");
  action = action.toUpperCase();

  if (action == "CLOSED") {
    Serial.println("============================================");
    Serial.println("  PERINTAH AI  >>  TUTUP  (tutup atap jemuran)");
    Serial.println("============================================");
    if (DRIVE_ACTUATORS_FROM_CAM) moveRoof("CLOSED");   // gerakkan aktuator TUTUP
  } else if (action == "OPENED" || action == "OPEN") {
    Serial.println("============================================");
    Serial.println("  PERINTAH AI  >>  BUKA   (buka atap jemuran)");
    Serial.println("============================================");
    if (DRIVE_ACTUATORS_FROM_CAM) moveRoof("OPEN");     // gerakkan aktuator BUKA
  } else {
    Serial.println("  PERINTAH AI  >>  TETAP  (tidak ada perubahan atap)");
  }
}

// Ekstraksi sederhana nilai string JSON: cari "key":"value"
String extractJsonField(String json, String key) {
  int idx = json.indexOf("\"" + key + "\"");
  if (idx < 0) return "";
  int colon = json.indexOf(':', idx);
  if (colon < 0) return "";
  int q1 = json.indexOf('"', colon);
  if (q1 < 0) return "";
  int q2 = json.indexOf('"', q1 + 1);
  if (q2 < 0) return "";
  return json.substring(q1 + 1, q2);
}

// ==========================================
// 6C. AKTUATOR ATAP: 2 penggerak buka & tutup
// ==========================================
void setupActuators() {
  pinMode(ACTUATOR_OPEN_PIN, OUTPUT);
  pinMode(ACTUATOR_CLOSE_PIN, OUTPUT);
  digitalWrite(ACTUATOR_OPEN_PIN, LOW);
  digitalWrite(ACTUATOR_CLOSE_PIN, LOW);
  Serial.println("[AKTUATOR] Pin aktuator atap siap | OPEN=GPIO" + String(ACTUATOR_OPEN_PIN) +
                 " CLOSE=GPIO" + String(ACTUATOR_CLOSE_PIN));
}

// Gerakkan atap melalui 2 aktuator (pulsa ROOF_MOVE_MS lalu matikan)
void moveRoof(const String& dir) {
  int pin = -1;
  if (dir == "CLOSED") {
    pin = ACTUATOR_CLOSE_PIN;
    Serial.println("  >> Mengaktifkan aktuator TUTUP...");
  } else if (dir == "OPEN" || dir == "OPENED") {
    pin = ACTUATOR_OPEN_PIN;
    Serial.println("  >> Mengaktifkan aktuator BUKA...");
  } else {
    return; // TETAP: tidak gerakkan aktuator
  }
  if (pin >= 0) {
    digitalWrite(pin, HIGH);
    delay(ROOF_MOVE_MS);
    digitalWrite(pin, LOW);
    Serial.println("  >> Aktuator selesai.");
  }
}

// ==========================================
// 7. KONEKSI WI-FI
// ==========================================
void connectWiFi() {
  Serial.printf("[WIFI] Menghubungkan ESP32-CAM ke: %s ", WIFI_SSID);
  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

  int tries = 0;
  while (WiFi.status() != WL_CONNECTED && tries < 20) {
    delay(500);
    Serial.print(".");
    tries++;
  }

  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("\n[WIFI] Terhubung! IP ESP32-CAM : " + WiFi.localIP().toString()); // <-- IP di Serial Monitor
    Serial.println("[WIFI] Server Upload Sky : " + SERVER_UPLOAD_URL);
    Serial.println("----------------------------------------------------------");
    Serial.println("  Pastikan IP pada SERVER_UPLOAD_URL sama dengan IP Laptop/XAMPP");
    Serial.println("  agar foto bisa diunggah ke dashboard SkyGuard AI.");
    Serial.println("----------------------------------------------------------");
  } else {
    Serial.println("\n[WIFI] Gagal terhubung ke Wi-Fi!");
  }
}
