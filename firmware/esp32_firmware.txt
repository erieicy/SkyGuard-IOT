   3. 2 Aktuator Atap Jemuran (Relay / Motor DC: satu untuk BUKA, satu untuk TUTUP).

#include <WiFi.h>
#include <HTTPClient.h>
#include <ArduinoJson.h>
#include <Preferences.h>

// ==========================================
// 1. PENGATURAN WI-FI & SERVER
// ==========================================
// GANTI dengan SSID dan Password Wi-Fi Anda yang nyala
const char* WIFI_SSID     = "GANTI_SSID_ANDA";
const char* WIFI_PASSWORD = "GANTI_PASSWORD_ANDA";

// URL Server SkyGuard AI — GANTI dengan IP laptop/XAMPP Anda
// Cek IP laptop: buka cmd, ketik "ipconfig", cari IPv4 Address
// Contoh: "http://192.168.1.100/SkyGuard-AI/api/esp32.php"
// Server ini: http://10.27.249.159/SkyGuard-AI/api/esp32.php
String serverUrl = "http://10.27.249.159/SkyGuard-AI/api/esp32.php";

// ==========================================
// 2. PINOUT HARDWARE ESP32
// ==========================================
#define PIN_RAIN_DIGITAL   34   // Sensor Hujan Digital Output (DO)
#define PIN_RAIN_ANALOG    35   // Sensor Hujan Analog Output (AO)
#define PIN_LDR_ANALOG     32   // Sensor Cahaya LDR Analog

// 2 AKTUATOR ATAP (buka & tutup) — TIDAK ADA servo / buzzer / LED.
#define ACTUATOR_OPEN_PIN   18  // GPIO penggerak atap MEMBUKA
#define ACTUATOR_CLOSE_PIN  19  // GPIO penggerak atap MENUTUP
// Durasi pulsa aktuator (ms) — sesuaikan dengan waktu buka/tutup fisik atap
const unsigned long ROOF_MOVE_MS = 3000;

// ==========================================
// 3. VARIABEL & OBJEK KONTROL
// ==========================================
// Penyimpanan persisten (EEPROM emulasi) untuk URL Server
Preferences preferences;

// Web server kecil di ESP32 untuk menerima perintah "Hubungkan" dari Dashboard
WiFiServer esp32Server(80);

String currentRoofStatus = "CLOSED";
String controlMode = "AUTO";

unsigned long lastSensorUpdate = 0;
const unsigned long SENSOR_INTERVAL = 3000; // Kirim data sensor setiap 3 detik

unsigned long lastCommandPoll = 0;
const unsigned long COMMAND_INTERVAL = 2000; // Cek perintah server setiap 2 detik

// ==========================================
// 4. SETUP SISTEM
// ==========================================
void setup() {
  Serial.begin(115200);
  delay(1000);
  Serial.println("\n========================================");
  Serial.println("  SKYGUARD AI - ESP32 CONTROLLER BOOTING ");
  Serial.println("========================================");

  // Inisialisasi Pin Sensor
  pinMode(PIN_RAIN_DIGITAL, INPUT);
  pinMode(PIN_RAIN_ANALOG, INPUT);
  pinMode(PIN_LDR_ANALOG, INPUT);

  // Inisialisasi 2 pin aktuator atap (buka & tutup)
  setupActuators();

  // Set awal atap tertutup untuk keamanan
  moveRoofActuator("CLOSED");

  // Hubungkan ke Wi-Fi
  connectToWiFi();

  // Muat URL Server dari memori persisten (jika pernah diatur via Dashboard "Hubungkan")
  preferences.begin("skyguard", false);
  String savedUrl = preferences.getString("server_url", "");
  if (savedUrl.length() > 0) {
    serverUrl = savedUrl;
    Serial.println("[CONFIG] URL Server dimuat dari memori: " + serverUrl);
  } else {
    Serial.println("[CONFIG] URL Server default: " + serverUrl);
  }

  // Mulai Web Server kecil di port 80 untuk menerima perintah dari Dashboard
  esp32Server.begin();
  Serial.println("[WEB] ESP32 HTTP Server aktif di port 80. Dashboard dapat mengirim /connect?server=URL");
}

// ==========================================
// 5. LOOP UTAMA
// ==========================================
void loop() {
  // Jaga koneksi Wi-Fi tetap aktif
  if (WiFi.status() != WL_CONNECTED) {
    connectToWiFi();
  }

  // Tangani perintah masuk dari Dashboard (misal: /connect?server=URL)
  handleEsp32Server();

  // 1. Baca Sensor Lokal
  int rainDigital = digitalRead(PIN_RAIN_DIGITAL); // 0 = Air terdeteksi (Active LOW pada modul umum)
  int rainAnalog  = analogRead(PIN_RAIN_ANALOG);   // Nilai 0 (basah) - 4095 (kering)
  int ldrAnalog   = analogRead(PIN_LDR_ANALOG);    // Nilai 0 (terang) - 4095 (gelap) — modul LDR terbalik!

  // Normalisasi: 1 = Hujan/Basah, 0 = Kering
  int isRainDetected = (rainDigital == LOW || rainAnalog < 2000) ? 1 : 0;
  
  // Konversi LDR ke persentase (0 - 100%) — INVERTED karena modul LDR terbalik
  int lightPercent = map(ldrAnalog, 0, 4095, 100, 0);
  lightPercent = constrain(lightPercent, 0, 100);

  // 2. Proteksi Darurat Lokal (Zero Latency Protection):
  // Jika air menyentuh sensor secara mendadak, langsung tutup atap tanpa menunggu server!
  if (isRainDetected == 1 && currentRoofStatus != "CLOSED") {
    Serial.println("\n[DARURAT] Air hujan terdeteksi! Menutup atap jemuran secara instan...");
    moveRoofActuator("CLOSED");
  }

  // 3. Kirim Telemetri Sensor ke Server
  if (millis() - lastSensorUpdate >= SENSOR_INTERVAL) {
    lastSensorUpdate = millis();
    sendSensorDataToServer(isRainDetected, lightPercent);
  }

  // 4. Cek Perintah dari Server (Manual Open/Close, Mode AI, Timer)
  if (millis() - lastCommandPoll >= COMMAND_INTERVAL) {
    lastCommandPoll = millis();
    fetchCommandsFromServer();
  }

  delay(50);
}

// ==========================================
// 6. FUNGSI GERAK 2 AKTUATOR ATAP JEMURAN
// ==========================================
void setupActuators() {
  pinMode(ACTUATOR_OPEN_PIN, OUTPUT);
  pinMode(ACTUATOR_CLOSE_PIN, OUTPUT);
  digitalWrite(ACTUATOR_OPEN_PIN, LOW);
  digitalWrite(ACTUATOR_CLOSE_PIN, LOW);
  Serial.println("[AKTUATOR] Pin aktuator atap siap | OPEN=GPIO" + String(ACTUATOR_OPEN_PIN) +
                 " CLOSE=GPIO" + String(ACTUATOR_CLOSE_PIN));
}

// Gerakkan atap lewat 2 aktuator (pulsa ROOF_MOVE_MS lalu matikan).
// dir: "OPEN" -> aktuator buka, "CLOSED" -> aktuator tutup.
void moveRoofActuator(const String& dir) {
  int pin = -1;
  if (dir == "CLOSED") {
    pin = ACTUATOR_CLOSE_PIN;
    Serial.println("[AKTUATOR] Menutup atap (pulsa aktuator TUTUP)...");
  } else if (dir == "OPEN") {
    pin = ACTUATOR_OPEN_PIN;
    Serial.println("[AKTUATOR] Membuka atap (pulsa aktuator BUKA)...");
  } else {
    return;
  }
  if (pin >= 0) {
    digitalWrite(pin, HIGH);
    delay(ROOF_MOVE_MS);
    digitalWrite(pin, LOW);
    currentRoofStatus = dir;
    Serial.println("[AKTUATOR] Selesai. Status atap: " + dir);
  }
}

// ==========================================
// 7. FUNGSI KOMUNIKASI HTTP REST API
// ==========================================
void sendSensorDataToServer(int rain, int light) {
  if (WiFi.status() != WL_CONNECTED) return;

  HTTPClient http;
  String url = serverUrl + "?action=update_sensors";
  http.begin(url);
  http.addHeader("Content-Type", "application/json");

  // Buat Payload JSON
  StaticJsonDocument<200> doc;
  doc["rain"] = rain;
  doc["light"] = light;
  doc["device_id"] = "esp32_main_controller";

  String requestBody;
  serializeJson(doc, requestBody);

  int httpCode = http.POST(requestBody);
  if (httpCode > 0) {
    String response = http.getString();
    // Serial.println("[HTTP SENSOR] Response: " + response);
  } else {
    Serial.printf("[HTTP SENSOR] Gagal kirim data, error: %s\n", http.errorToString(httpCode).c_str());
  }
  http.end();
}

void fetchCommandsFromServer() {
  if (WiFi.status() != WL_CONNECTED) return;

  HTTPClient http;
  String url = serverUrl + "?action=get_command";
  http.begin(url);

  int httpCode = http.GET();
  if (httpCode == HTTP_CODE_OK) {
    String payload = http.getString();
    
    StaticJsonDocument<512> doc;
    DeserializationError error = deserializeJson(doc, payload);
    if (!error) {
      String desiredRoofStatus = doc["roof_status"] | "CLOSED";
      String mode = doc["control_mode"] | "AUTO";
      int rainDetected = doc["rain_detected"] | 0;

      controlMode = mode;

      // Eksekusi jika ada perbedaan status
      if (desiredRoofStatus != currentRoofStatus) {
        Serial.printf("[SERVER CMD] Perintah diterima: %s (Mode: %s)\n", desiredRoofStatus.c_str(), mode.c_str());
        
        if (desiredRoofStatus == "OPEN" && rainDetected == 0) {
          moveRoofActuator("OPEN");
        } else {
          moveRoofActuator("CLOSED");
        }
      }
    }
  }
  http.end();
}

// ==========================================
// 7B. WEB SERVER KECIL (Terima Perintah "Hubungkan" dari Dashboard)
// ==========================================
void handleEsp32Server() {
  WiFiClient client = esp32Server.available();
  if (!client) return;

  // Baca baris request pertama: "GET /connect?server=... HTTP/1.1"
  String reqLine = client.readStringUntil('\n');
  reqLine.trim();

  // Ambil path + query string
  int firstSpace = reqLine.indexOf(' ');
  int secondSpace = reqLine.indexOf(' ', firstSpace + 1);
  String path = (firstSpace >= 0 && secondSpace >= 0)
      ? reqLine.substring(firstSpace + 1, secondSpace)
      : "";

  // Baca sisa header hingga baris kosong
  while (client.available()) {
    String line = client.readStringUntil('\n');
    if (line == "\r" || line == "\n" || line == "") break;
  }

  String body = "SkyGuard ESP32 Controller Ready";

  if (path.startsWith("/connect")) {
    // Parse query: ?server=URL_ENCODED
    int q = path.indexOf("server=");
    if (q >= 0) {
      String newUrl = path.substring(q + 7);

      // URL-decode untuk karakter URL umum
      newUrl.replace("%2F", "/");
      newUrl.replace("%2f", "/");
      newUrl.replace("%3A", ":");
      newUrl.replace("%3a", ":");
      newUrl.replace("%3D", "=");
      newUrl.replace("%3d", "=");
      newUrl.replace("%26", "&");
      newUrl.replace("%3F", "?");
      newUrl.replace("%3f", "?");
      newUrl.replace("+", " ");

      serverUrl = newUrl;
      preferences.putString("server_url", serverUrl);
      Serial.println("\n[WEB] Perintah 'Hubungkan' diterima dari Dashboard.");
      Serial.println("[WEB] ESP32 akan mempolling server: " + serverUrl);
      body = "SKYGUARD_OK ESP32 Controller mempolling: " + serverUrl;
    } else {
      body = "ERROR: parameter 'server' tidak ditemukan";
    }
  }

  // Kirim respons HTTP sederhana
  String resp = "HTTP/1.1 200 OK\r\n";
  resp += "Content-Type: text/plain\r\n";
  resp += "Connection: close\r\n\r\n";
  resp += body;
  client.print(resp);
  client.stop();
}

// ==========================================
// 8. KONEKSI WI-FI
// ==========================================
void connectToWiFi() {
  Serial.printf("\n[WIFI] Menghubungkan ke: %s ", WIFI_SSID);
  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

  int retry = 0;
  while (WiFi.status() != WL_CONNECTED && retry < 25) {
    delay(500);
    Serial.print(".");
    retry++;
  }

  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("\n[WIFI] Terhubung Berhasil!");
    Serial.print("[WIFI] IP Address ESP32 : ");
    Serial.println(WiFi.localIP());          // <-- IP ini yg muncul di Serial Monitor
    Serial.print("[WIFI] Server SkyGuard  : ");
    Serial.println(serverUrl);               // <-- URL XAMPP yg harus dituju ESP32
    Serial.println("----------------------------------------------------------");
    Serial.println("  Buka Serial Monitor untuk melihat IP di atas.");
    Serial.println("  Pastikan serverUrl di atas sudah sama dengan IP Laptop/XAMPP.");
    Serial.println("----------------------------------------------------------");
  } else {
    Serial.println("\n[WIFI] Gagal terhubung ke Wi-Fi. Akan mencoba lagi...");
  }
}
